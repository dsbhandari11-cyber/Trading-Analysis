//@version=5
strategy("RSI Strategy with Dynamic Thresholds + Sell Pressure Filter",
     shorttitle="RSI Strategy DJ", overlay=true)

//--------------------------------------------------------
// Core Parameters
//--------------------------------------------------------
rsi_period          = input.int(19, title="RSI Period")
overbought_base     = input.int(72, title="Base Overbought Level")
oversold_base       = input.int(31, title="Base Oversold Level")
ma_period           = input.int(46, title="Moving Average Period")
use_ma_confirmation = input.bool(true, title="Use Moving Average Confirmation")

// Volatility-based adjustment
atr_period  = input.int(14, title="ATR Period")
atr         = ta.atr(atr_period)
adjustment  = atr * 0.02  // Further reduced adjustment factor

overbought  = overbought_base + adjustment
oversold    = oversold_base - adjustment

//--------------------------------------------------------
// Indicators used in filters
//--------------------------------------------------------
// RSI
rsi = ta.rsi(close, rsi_period)

// Moving average
ma  = ta.sma(close, ma_period)

// ROC (for option 1)
roc_length = input.int(9, title="ROC Length", minval=1)
roc        = ta.roc(close, roc_length)

// MACD (for option 3)
macd_fast   = input.int(12, "MACD Fast Length")
macd_slow   = input.int(26, "MACD Slow Length")
macd_signal = input.int(9, "MACD Signal Length")
[macd_line, macd_signal_line, macd_hist] = ta.macd(close, macd_fast, macd_slow, macd_signal)

// Volume selling (for option 4)
vol_len = input.int(20, "Volume SMA Length")
vol_sma = ta.sma(volume, vol_len)

// ADX / DMI (for option 5)
// NOTE: correct use is ta.dmi(high, low, close, length)
//--------------------------------------------------------
// ADX / DMI (for option 5)
//--------------------------------------------------------
dmi_len       = input.int(14, "DMI Length")
adx_smoothing = input.int(14, "ADX Smoothing")

// Correct: Only 2 arguments allowed
[diPlus, diMinus, adx] = ta.dmi(dmi_len, adx_smoothing)

//--------------------------------------------------------
// Sell Pressure Filter Selection
//--------------------------------------------------------
use_sell_filter = input.bool(true, "Use Sell Pressure Filter?")
sell_filter_type = input.string(     "ROC",     "Sell Pressure Filter Type",  options = ["ROC", "RSI < 50", "MACD < 0", "Volume Selling", "ADX DI- > DI+", "None"])

// Build sell pressure condition based on selected option
sellPressure =     sell_filter_type == "ROC"            ? roc < 0 : sell_filter_type == "RSI < 50"       ? rsi < 50 :
     sell_filter_type == "MACD < 0"       ? macd_hist < 0 :     sell_filter_type == "Volume Selling" ? (volume > vol_sma and close < open) :
     sell_filter_type == "ADX DI- > DI+"  ? (diMinus > diPlus and adx > 20) :
     true  // "None": no extra sell pressure condition

//--------------------------------------------------------
// Base Signals (RSI + MA logic)
//--------------------------------------------------------
rawBuy =
     ta.crossover(rsi, oversold) or
     (rsi < oversold and (not use_ma_confirmation or ta.crossover(close, ma)))

rawSell =
     ta.crossunder(rsi, overbought) or
     (rsi > overbought and (not use_ma_confirmation or ta.crossunder(close, ma)))

// Apply sell pressure filter only to sell signals
buySignal  = rawBuy
sellSignal = rawSell and (not use_sell_filter or sellPressure)

//--------------------------------------------------------
// Orders
//--------------------------------------------------------
if buySignal
    strategy.entry("Buy", strategy.long)

if sellSignal
    strategy.entry("Sell", strategy.short)

//--------------------------------------------------------
// Candle / background visualization
//--------------------------------------------------------
use_candle_colors = input.bool(true, "Color Candles by Pressure / Signals?")
use_bg_shading    = input.bool(true, "Shade Background in Sell Zone?")

// Define zones
sellZone = use_sell_filter and sellPressure

// Candle colors:
//  - Red when in sell zone
//  - Lime on bar where a buy signal appears
//  - Otherwise default chart color
candleColor =
     not use_candle_colors ? na :
     sellZone              ? color.new(color.red, 0) :
     buySignal             ? color.new(color.lime, 0) :
     na

barcolor(candleColor)

// Optional red background when sell pressure is active
bgcolor(use_bg_shading and sellZone ? color.new(color.maroon, 92) : na)

//--------------------------------------------------------
// Plotting labels & debug
//--------------------------------------------------------
plotshape(     series=buySignal, title="Buy Signal", location=location.belowbar,     color=color.rgb(8, 128, 240),  style=shape.labelup,     text="Buy")

plotshape(     series=sellSignal, title="Sell Signal",     location=location.abovebar,     color=color.rgb(8, 0, 0), style=shape.labeldown,     text="Sell")

// ATR Debug
plot(adjustment, title="ATR Adjustment", color=color.rgb(9, 6, 199))
